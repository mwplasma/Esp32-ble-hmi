#include #include #include // ======================================== // Motor pins // ======================================== #define PUL_PIN 25 #define DIR_PIN 27 // ======================================== // BLE UUID // ======================================== #define SERVICE_UUID \ "12345678-1234-1234-1234-123456789abc" #define CHARACTERISTIC_UUID \ "abcd1234-1234-1234-1234-abcdef123456" // ======================================== // Motor / Ball Screw settings // ======================================== // Verified driver setting const int PULSES_PER_REV = 800; // Ball screw lead const float MM_PER_REV = 5.0; // 800 pulses/rev / 5 mm/rev = 160 pulses/mm const float PULSES_PER_MM = PULSES_PER_REV / MM_PER_REV; // ======================================== // JOG speed settings // ======================================== float motorSpeedMMs = 10.0; const float MIN_SPEED_MMS = 5.0; const float MAX_SPEED_MMS = 50.0; // ======================================== // Motor state // ======================================== bool motorRunning = false; bool pulseState = HIGH; unsigned long pulseInterval = 0; unsigned long lastPulseTime = 0; // BLE state bool deviceConnected = false; // ======================================== // Update pulse interval from mm/s // ======================================== void updatePulseInterval() { float pulsesPerSecond = PULSES_PER_MM * motorSpeedMMs; // Output toggles HIGH/LOW each interval, // therefore use half of 1 second. pulseInterval = (unsigned long)( 500000.0 / pulsesPerSecond ); Serial.print("Speed: "); Serial.print(motorSpeedMMs, 1); Serial.println(" mm/s"); Serial.print("Pulse Rate: "); Serial.print(pulsesPerSecond, 1); Serial.println(" pulses/sec"); Serial.print("Pulse Interval: "); Serial.print(pulseInterval); Serial.println(" us"); } // ======================================== // Set JOG speed in mm/s // ======================================== void setSpeedMMs(float speedMMs) { if (speedMMs < MIN_SPEED_MMS) speedMMs = MIN_SPEED_MMS; if (speedMMs > MAX_SPEED_MMS) speedMMs = MAX_SPEED_MMS; motorSpeedMMs = speedMMs; updatePulseInterval(); } // ======================================== // Direction control // ======================================== void setCW() { // DIR- -> GND pinMode(DIR_PIN, OUTPUT); digitalWrite(DIR_PIN, LOW); Serial.println("Direction: CW"); } void setCCW() { // DIR- -> OPEN pinMode(DIR_PIN, INPUT); Serial.println("Direction: CCW"); } // ======================================== // Motor control // ======================================== void startMotor() { lastPulseTime = micros(); motorRunning = true; Serial.println("Motor: RUN"); } void stopMotor() { motorRunning = false; pulseState = HIGH; digitalWrite(PUL_PIN, HIGH); Serial.println("Motor: STOP"); } // ======================================== // BLE command processing // ======================================== void processCommand(String command) { command.trim(); Serial.print("BLE Command: "); Serial.println(command); // Forward / CW if (command == "F") { stopMotor(); setCW(); delay(100); startMotor(); } // Reverse / CCW else if (command == "R") { stopMotor(); setCCW(); delay(100); startMotor(); } // Stop else if (command == "S") { stopMotor(); } // Speed in mm/s // V5, V10, V15 ... V50 else if (command.startsWith("V")) { float speedMMs = command.substring(1).toFloat(); if (speedMMs >= MIN_SPEED_MMS && speedMMs <= MAX_SPEED_MMS) { setSpeedMMs(speedMMs); } } } // ======================================== // BLE characteristic callback // ======================================== class CommandCallback : public BLECharacteristicCallbacks { void onWrite( BLECharacteristic *pCharacteristic) { String command = pCharacteristic->getValue().c_str(); processCommand(command); } }; // ======================================== // BLE SERVER CALLBACK // Handles connect / disconnect // ======================================== class ServerCallbacks : public BLEServerCallbacks { void onConnect(BLEServer *pServer) { deviceConnected = true; Serial.println(); Serial.println("BLE CONNECTED"); } void onDisconnect(BLEServer *pServer) { deviceConnected = false; Serial.println(); Serial.println("BLE DISCONNECTED"); // Safety: // Stop motor if communication is lost stopMotor(); // Start advertising again delay(200); pServer->startAdvertising(); Serial.println( "BLE Advertising Restarted" ); Serial.println( "Waiting for Galaxy Tab..." ); } }; // ======================================== // SETUP // ======================================== void setup() { Serial.begin(115200); // IMPORTANT: // Keep verified motor output configuration pinMode(PUL_PIN, OUTPUT_OPEN_DRAIN); digitalWrite(PUL_PIN, HIGH); setCW(); updatePulseInterval(); // ---------------------------------------- // BLE // ---------------------------------------- BLEDevice::init("Project_Rebirth"); BLEServer *pServer = BLEDevice::createServer(); // Connect / disconnect callback pServer->setCallbacks( new ServerCallbacks() ); BLEService *pService = pServer->createService( SERVICE_UUID ); BLECharacteristic *pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_WRITE ); pCharacteristic->setCallbacks( new CommandCallback() ); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID( SERVICE_UUID ); pAdvertising->setScanResponse(true); BLEDevice::startAdvertising(); Serial.println(); Serial.println( "Project Rebirth BLE Ready" ); Serial.println( "Waiting for Galaxy Tab..." ); } // ======================================== // MAIN LOOP // ======================================== void loop() { if (motorRunning) { unsigned long currentTime = micros(); if (currentTime - lastPulseTime >= pulseInterval) { lastPulseTime = currentTime; pulseState = !pulseState; digitalWrite( PUL_PIN, pulseState ); } } }
